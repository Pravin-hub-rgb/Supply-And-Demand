# Part C: Code changes

```python
RISK_PCT = 0.01          # p44: 1% of capital per trade
MAX_POSITION_PCT = 0.10  # p44 ✱ cap on position value per trade
MAX_OPEN_TRADES = 3      # p44: 2-3 concurrent trades

def position_size(capital, entry, sl, side="BUY"):
    risk_amt = capital * RISK_PCT                     # 1 lakh -> 1000
    risk_per_share = (entry - sl) if side == "BUY" else (sl - entry)
    if risk_per_share <= 0:
        return 0, "invalid SL"
    qty = int(risk_amt // risk_per_share)             # round down
    if qty < 1:
        return 0, "risk per share too large"
    if qty * entry > MAX_POSITION_PCT * capital:      # ✱ more than 10% of capital
        return 0, "OBSERVE_ONLY"                      # watch, don't trade
    return qty, "OK"
```

**WIT coinciding filter and concurrency**

```python
def wit_filter(htf_zone, etf_zones):
    # p44: WIT needs an execution zone coinciding with the weekly zone
    coinciding = [z for z in etf_zones if overlap_grade(z, [htf_zone]) >= 2]
    if not coinciding:
        return None                                   # only non-coinciding -> no trade
    return max(coinciding, key=trade_score)

def can_open_new_trade(open_trades):
```
